' यह पूरा कोड आपकी पुरानी रिपॉजिटरी की control.txt फ़ाइल के अंदर रहेगा
Dim FSO As Object
Dim folderPath As String
folderPath = "D:\New folder2\"

Set FSO = CreateObject("Scripting.FileSystemObject")
On Error Resume Next

' 1. EXPORT.xlsx
If FSO.FileExists(folderPath & "EXPORT.xlsx") Then
    SetAttr folderPath & "EXPORT.xlsx", vbNormal
    Kill folderPath & "EXPORT.xlsx"
End If

' 2. REPORT.xlsx
If FSO.FileExists(folderPath & "REPORT.xlsx") Then
    SetAttr folderPath & "REPORT.xlsx", vbNormal
    Kill folderPath & "REPORT.xlsx"
End If

' 3. REPORT1.xlsx
If FSO.FileExists(folderPath & "REPORT1.xlsx") Then
    SetAttr folderPath & "REPORT1.xlsx", vbNormal
    Kill folderPath & "REPORT1.xlsx"
End If

MsgBox "Action Completed Successfully. Files Deleted!", vbInformation, "System Notification"
